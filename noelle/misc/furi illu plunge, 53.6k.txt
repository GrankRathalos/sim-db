# Options
options iteration=100; 
options swap_delay=12;
energy every interval=480,720 amount=1;

# Targets
target lvl=100 resist=0.1 radius=2 pos=0,2.4 hp=100000000;
#target lvl=100 resist=0.1 radius=2 pos=3.4,0 hp=100000000; 

# Rotation
active noelle; # SS Stacks

for let i = 0; i < 4; i = i + 1 {

  furina skill, dash, burst;
  while !.furina.mods.favonius-cd {
    furina attack;
  }
  
  xianyun skill, low_plunge, attack, burst;
  
  illuga burst, skill;
  
  noelle swap;
  while .xianyun.status.xianyun-airborne-buff {
    if !.noelle.mods.noelle-burst {
      noelle burst;
    }
    if .xianyun.status.xianyun-airborne-buff {
      noelle jump;
      if .airborne {
        noelle high_plunge[collision=rand()>0.5], dash;
      }
    }
  }
  illuga attack:3;
  while .furina.skill.cd {
    if .noelle.skill.ready {
      noelle skill;
    }
    if .noelle.normal > 3 {
      noelle dash;
    } else {
      noelle attack;
    }
  }
}
